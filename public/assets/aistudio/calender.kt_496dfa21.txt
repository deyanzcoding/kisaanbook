// Province-specific notes for important crops
fun getProvinceCropNote(crop: String, province: Province, monthId: Int): String {
    val c = crop.lowercase()
    return when {
        c.contains("wheat") -> when (province) {
            Province.SINDH -> if (monthId == 10 || monthId == 11) "Lower Sindh: best 1–20 Nov. Upper Sindh: from 7 Nov."
            Province.KP -> if (monthId == 10) "Early sowing from ~25 Oct recommended."
            Province.PUNJAB -> "Main window Oct–Nov. Aim for early November."
            else -> "Oct–Dec depending on area."
        }
        c.contains("cotton") -> when (province) {
            Province.SINDH -> "Earlier start possible (Feb–Mar in south)."
            else -> "Main sowing April–May."
        }
        c.contains("rice") -> when (province) {
            Province.SINDH -> "Nursery earlier (April–May)."
            else -> "Nursery May–June, transplant June–July."
        }
        else -> ""
    }
}

// ========== LOCATION HELPER ==========

@SuppressLint("MissingPermission")
fun detectProvince(lat: Double, lon: Double): Province {
    // Approximate bounding boxes (good enough for province-level)
    return when {
        lat in 23.5..28.5 && lon in 66.5..71.5 -> Province.SINDH
        lat in 27.5..34.0 && lon in 69.5..75.5 -> Province.PUNJAB
        lat in 31.5..37.0 && lon in 69.0..74.5 -> Province.KP
        lat in 24.5..32.0 && lon in 60.5..70.5 -> Province.BALOCHISTAN
        lat in 33.4..33.9 && lon in 72.8..73.3 -> Province.ISLAMABAD
        else -> Province.UNKNOWN
    }
}

// ========== MAIN SCREEN ==========

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun CalendarScreen() {
    val context = LocalContext.current
    var selectedType by remember { mutableStateOf(CalendarType.GREGORIAN) }
    var selectedMonth by remember { mutableStateOf<MonthInfo?>(null) }
    var province by remember { mutableStateOf(Province.UNKNOWN) }
    var searchQuery by remember { mutableStateOf("") }
    var locationStatus by remember { mutableStateOf("Detecting location…") }

    val today = LocalDate.now()
    val currentMonthId = today.monthValue

    // Permission launcher
    val permissionLauncher = rememberLauncherForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { permissions ->
        if (permissions[Manifest.permission.ACCESS_FINE_LOCATION] == true ||
            permissions[Manifest.permission.ACCESS_COARSE_LOCATION] == true) {
            fetchLocation(context) { lat, lon ->
                province = detectProvince(lat, lon)
                locationStatus = "Location: ${province.displayName}"
            }
        } else {
            locationStatus = "Location permission denied – using general Pakistan advice"
            province = Province.UNKNOWN
        }
    }

    // Request location on first load
    LaunchedEffect(Unit) {
        val fine = ContextCompat.checkSelfPermission(context, Manifest.permission.ACCESS_FINE_LOCATION)
        val coarse = ContextCompat.checkSelfPermission(context, Manifest.permission.ACCESS_COARSE_LOCATION)
        if (fine == PackageManager.PERMISSION_GRANTED || coarse == PackageManager.PERMISSION_GRANTED) {
            fetchLocation(context) { lat, lon ->
                province = detectProvince(lat, lon)
                locationStatus = "Location: ${province.displayName}"
            }
        } else {
            permissionLauncher.launch(
                arrayOf(
                    Manifest.permission.ACCESS_FINE_LOCATION,
                    Manifest.permission.ACCESS_COARSE_LOCATION
                )
            )
        }
    }

    // Filtered months for search
    val filteredMonths = if (searchQuery.isBlank()) {
        allMonths
    } else {
        allMonths.filter { month ->
            month.baseCrops.any { it.contains(searchQuery, ignoreCase = true) } ||
            month.gregorianName.contains(searchQuery, ignoreCase = true) ||
            month.pashtoName.contains(searchQuery, ignoreCase = true)
        }
    }

    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text("Pakistan Crop Calendar") },
                actions = {
                    var expanded by remember { mutableStateOf(false) }
                    TextButton(onClick = { expanded = true }) {
                        Text(selectedType.title, fontSize = 13.sp)
                    }
                    DropdownMenu(expanded = expanded, onDismissRequest = { expanded = false }) {
                        CalendarType.values().forEach { type ->
                            DropdownMenuItem(
                                text = { Text(type.title) },
                                onClick = {
                                    selectedType = type
                                    expanded = false
                                }
                            )
                        }
                    }
                }
            )
        }
    ) { padding ->
        Column(
            modifier = Modifier
                .padding(padding)
                .fillMaxSize()
                .padding(12.dp)
        ) {
            // Location status
            Text(
                text = locationStatus,
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.primary,
                modifier = Modifier.padding(bottom = 8.dp)
            )

            // Search bar
            OutlinedTextField(
                value = searchQuery,
                onValueChange = { searchQuery = it },
                modifier = Modifier.fillMaxWidth(),
                placeholder = { Text("Search crop (wheat, cotton, spinach…)") },
                leadingIcon = { Icon(Icons.Default.Search, contentDescription = null) },
                singleLine = true
            )

            Spacer(Modifier.height(12.dp))

            if (searchQuery.isNotBlank()) {
                Text(
                    "Showing months related to \"$searchQuery\"",
                    style = MaterialTheme.typography.labelLarge,
                    modifier = Modifier.padding(bottom = 8.dp)
                )
            }

            // Month grid
            LazyVerticalGrid(
                columns = GridCells.Fixed(3),
                verticalArrangement = Arrangement.spacedBy(10.dp),
                horizontalArrangement = Arrangement.spacedBy(10.dp),
                modifier = Modifier.weight(1f)
            ) {
                items(filteredMonths) { month ->
                    MonthCard(
                        month = month,
                        displayName = getDisplayName(month, selectedType),
                        isCurrent = month.id == currentMonthId,
                        onClick = { selectedMonth = month }
                    )
                }
            }
        }
    }

    // Month detail dialog
    selectedMonth?.let { month ->
        AlertDialog(
            onDismissRequest = { selectedMonth = null },
            title = {
                Column {
                    Text(getDisplayName(month, selectedType), fontWeight = FontWeight.Bold)
                    Text(
                        "${month.gregorianName} • ${province.displayName}",
                        style = MaterialTheme.typography.bodySmall
                    )
                }
            },
            text = {
                Column(modifier = Modifier.verticalScroll(rememberScrollState())) {
                    Text("Recommended activities / crops:", fontWeight = FontWeight.SemiBold)
                    Spacer(Modifier.height(8.dp))

                    month.baseCrops.forEach { crop ->
                        val note = getProvinceCropNote(crop, province, month.id)
                        Text("• $crop", fontWeight = FontWeight.Medium)
                        if (note.isNotEmpty()) {
                            Text("  → $note", fontSize = 13.sp, color = MaterialTheme.colorScheme.primary)
                        }
                        Spacer(Modifier.height(4.dp))
                    }

                    Spacer(Modifier.height(12.dp))
                    Text(
                        "Tip: Always confirm with your local Agriculture Extension office. Dates can shift 1–2 weeks by district and weather.",
                        fontSize = 12.sp,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                }
            },
            confirmButton = {
                TextButton(onClick = { selectedMonth = null }) {
                    Text("Close")
                }
            }
        )
    }
}

@Composable
fun MonthCard(
    month: MonthInfo,
    displayName: String,
    isCurrent: Boolean,
    onClick: () -> Unit
) {
    Card(
        modifier = Modifier
            .fillMaxWidth()
            .height(90.dp)
            .clickable { onClick() },
        colors = CardDefaults.cardColors(
            containerColor = if (isCurrent)
                MaterialTheme.colorScheme.primaryContainer
            else
                MaterialTheme.colorScheme.surfaceVariant
        )
    ) {
        Box(contentAlignment = Alignment.Center, modifier = Modifier.fillMaxSize().padding(6.dp)) {
            Text(
                text = displayName,
                textAlign = TextAlign.Center,
                fontWeight = if (isCurrent) FontWeight.Bold else FontWeight.Normal,
                fontSize = 13.sp
            )
        }
    }
}

fun getDisplayName(month: MonthInfo, type: CalendarType): String = when (type) {
    CalendarType.GREGORIAN -> month.gregorianName
    CalendarType.PASHTUN -> "${month.pashtoScript}\n${month.pashtoName}"
    CalendarType.HIJRI -> month.hijriName
}

@SuppressLint("MissingPermission")
fun fetchLocation(context: android.content.Context, onResult: (Double, Double) -> Unit) {
    val client = LocationServices.getFusedLocationProviderClient(context)
    client.getCurrentLocation(Priority.PRIORITY_BALANCED_POWER_ACCURACY, CancellationTokenSource().token)
        .addOnSuccessListener { location ->
            if (location != null) {
                onResult(location.latitude, location.longitude)
            }
        }
        .addOnFailureListener {
            // fallback – try last known
            client.lastLocation.addOnSuccessListener { loc ->
                if (loc != null) onResult(loc.latitude, loc.longitude)
            }
        }
}
